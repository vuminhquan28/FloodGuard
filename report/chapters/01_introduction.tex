\chapter{Giới Thiệu Đề Tài \& Phân Tích Yêu Cầu}

\section{Đặt Vấn Đề}
Hiện tượng ngập lụt đô thị và sạt lở, tràn đê vào mùa mưa bão đang là một trong những thách thức nghiêm trọng nhất tại các đô thị lớn ở Việt Nam cũng như trên thế giới. Tình trạng này gây thiệt hại nặng nề về kinh tế, tài sản, cản trở giao thông và tiềm ẩn nhiều rủi ro về sức khỏe, tính mạng con người.

Các giải pháp truyền thống như điều tiết thủ công, đo đạc bằng mắt thường hoặc vận hành trạm bơm tại chỗ bộc lộ nhiều nhược điểm:
\begin{itemize}
    \item \textbf{Độ trễ phản ứng cao:} Khi nước dâng nhanh đột ngột vào ban đêm, con người không kịp có mặt để vận hành máy bơm hay mở cống xả.
    \item \textbf{Thiếu thông tin cảnh báo sớm:} Người dân trong khu vực không được cập nhật kịp thời về tốc độ và mức độ nước dâng.
    \item \textbf{Nguy cơ hỏng hóc thiết bị:} Việc bật/tắt thiết bị công suất lớn liên tục do mực nước dập dềnh gây giảm tuổi thọ của bơm và van xả.
\end{itemize}

\section{Mục Tiêu Dự Án FloodGuard}
Dự án \textbf{FloodGuard} hướng tới việc xây dựng một hệ thống IoT giám sát và chống ngập tự động, nhỏ gọn, hoạt động bền bỉ và phản ứng tức thời. Hệ thống tập trung vào các mục tiêu then chốt:
\begin{enumerate}
    \item Tự động đo lường mực nước và lượng mưa liên tục theo thời gian thực với độ chính xác cao.
    \item Tự động ra quyết định kích hoạt máy bơm nước và góc mở van thoát lũ theo thuật toán máy trạng thái hữu hạn (\textit{Finite State Machine - FSM}) kết hợp dải trễ (\textit{Hysteresis}).
    \item Phát tín hiệu cảnh báo đa tầng thông qua hệ thống đèn LED trạng thái và còi hú báo động âm lượng lớn.
    \item Cung cấp giao diện Web Dashboard trực quan, cho phép người quản lý theo dõi và can thiệp điều khiển thủ công (\textit{Manual Override}) từ xa qua mạng Wi-Fi.
\end{enumerate}

\section{Phân Tích Yêu Cầu Chức Năng \& Mô Hình Hóa Use Case}

\subsection{Phân Định Ranh Giới Hệ Thống \& Tác Nhân Theo Chuẩn UML 2.5}
Trong mô hình hóa Use Case cho hệ thống nhúng và IoT theo chuẩn quốc tế UML 2.5:
\begin{itemize}
    \item \textbf{Ranh giới hệ thống (\textit{System Boundary}):} Là khối điều khiển phần mềm chạy trên vi điều khiển ESP32 (\textit{FloodGuard Firmware Controller}). Hệ thống chịu trách nhiệm thu nhận dữ liệu, tính toán thuật toán FSM/Hysteresis, kích hoạt thiết bị ngoại vi và phục vụ giao tiếp Web API.
    \item \textbf{Tác nhân (\textit{Actors}):} Là các thực thể bên ngoài tương tác trực tiếp với ranh giới hệ thống, bao gồm con người, bộ định thời và các thiết bị ngoại vi:
    \begin{enumerate}
        \item \textbf{Tác nhân chính (\textit{Primary Actors}):}
        \begin{itemize}
            \item \textbf{Người vận hành (\textit{Human Actor}):} Giám sát viên tương tác từ xa qua trình duyệt Web Dashboard để theo dõi telemetry, chuyển đổi chế độ vận hành (AUTO/MANUAL) và ra lệnh can thiệp cưỡng chế thiết bị.
            \item \textbf{Bộ định thời Timer / Watchdog (\textit{Internal Timer}):} Kích hoạt định kỳ chu kỳ đo cảm biến, chu kỳ tính toán FSM và đếm thời gian giám sát mất kết nối mạng để tự động kích hoạt chế độ an toàn Failsafe.
        \end{itemize}
        \item \textbf{Tác nhân phụ / Thiết bị ngoại vi (\textit{Secondary Actors / Devices}):}
        \begin{itemize}
            \item \textbf{\texttt{«device»} Cảm biến siêu âm HC-SR04 \& Cảm biến mưa FC-37:} Cung cấp tín hiệu xung Echo đo khoảng cách và điện áp tương tự AO/mức logic DO phản ánh lượng mưa.
            \item \textbf{\texttt{«device»} Module Relay 5V \& Máy bơm mini:} Cơ cấu chấp hành nhận tín hiệu kích tiếp điểm đóng/ngắt nguồn cho máy bơm thoát nước.
            \item \textbf{\texttt{«device»} Động cơ Servo SG90:} Cơ cấu chấp hành nhận xung PWM điều khiển góc mở cửa van thoát lũ ($0^\circ, 90^\circ, 180^\circ$).
            \item \textbf{\texttt{«device»} Cụm 3 Đèn LED \& Còi Buzzer 5V:} Cơ cấu chỉ thị nhận tín hiệu điều khiển mức logic và xung âm thanh báo động tại trạm.
            \item \textbf{Người dân quanh trạm (\textit{Human Actor}):} Thụ hưởng và tiếp nhận trực tiếp tín hiệu cảnh báo bằng âm thanh còi hú và ánh sáng đèn LED tại thực địa.
        \end{itemize}
    \end{enumerate}
\end{itemize}

\subsubsection{UC01: Đọc và Tiền Xử Lý Dữ Liệu Cảm Biến}
\begin{itemize}
    \item \textbf{Mã định danh:} \texttt{UC01}
    \item \textbf{Tác nhân:}
    \begin{itemize}
        \item \textit{Tác nhân chính:} Bộ định thời Timer 100ms (kích hoạt chu kỳ đọc).
        \item \textit{Tác nhân phụ:} \texttt{«device»} Cảm biến siêu âm HC-SR04, \texttt{«device»} Cảm biến mưa FC-37.
    \end{itemize}
    \item \textbf{Mục tiêu:} Thu thập dữ liệu cảm biến sạch, triệt tiêu xung gai nhiễu điện từ (EMI) và quy đổi ra \% mực nước cùng các cấp độ mưa.
    \item \textbf{Tiền điều kiện:} Cảm biến được cấp nguồn ổn định; chân Trig/Echo và AO/DO nối đúng sơ đồ chân GPIO.
    \item \textbf{Quan hệ \texttt{«include»} (chuẩn UML 2.5):} UC01 trực tiếp gọi đồng thời 3 ca sử dụng con (không tạo chuỗi tuần tự):
    \begin{enumerate}
        \item \textit{Đo khoảng cách siêu âm (HC-SR04):} Kích xung Trig 10$\mu$s, đo độ rộng xung Echo.
        \item \textit{Lọc trung vị 3 mẫu \& Tính \% nước:} Loại trừ mẫu ngoại lai, quy đổi khoảng cách sang dải 0\% -- 100\%.
        \item \textit{Đọc tín hiệu mưa \& Phân cấp ADC/DO:} Đọc mức ADC và phân loại 3 mức độ (mưa nhỏ, mưa to, ngập nước).
    \end{enumerate}
    \item \textbf{Hậu điều kiện:} Cung cấp các thông số chuẩn hóa \texttt{waterPercent}, \texttt{currentDistance}, \texttt{rainRawADC}, \texttt{isRaining}, \texttt{isHeavyRain}, \texttt{isSubmerged} cho thuật toán FSM và Web Server.
\end{itemize}

\subsubsection{UC02: Tự Động Đánh Giá Ngưỡng Trạng Thái \& Xử Lý Chống Rung (Hysteresis)}
\begin{itemize}
    \item \textbf{Mã định danh:} \texttt{UC02}
    \item \textbf{Tác nhân chính:} Bộ định thời chu kỳ FSM (kích hoạt định kỳ đánh giá logic).
    \item \textbf{Mục tiêu:} Xác định trạng thái vận hành của trạm và triệt tiêu hiện tượng rung giật đóng/ngắt thiết bị liên tục (chattering/jittering) khi mặt nước dao động quanh biên ngưỡng.
    \item \textbf{Tiền điều kiện:} Hệ thống ở chế độ \texttt{AUTO MODE} và dữ liệu từ \texttt{UC01} đã sẵn sàng.
    \item \textbf{Quan hệ \texttt{«include»}:} UC02 trực tiếp gọi:
    \begin{itemize}
        \item \textit{Đánh giá 4 cấp độ (NORMAL -- CRITICAL):} Đối chiếu \% nước và cấp mưa với bảng quy tắc ngưỡng.
        \item \textit{Áp dụng dải trễ Hysteresis chống rung:} Duy trì xả cho tới khi nước rút sâu $< 40\%$ mới hạ trạng thái.
    \end{itemize}
    \item \textbf{Hậu điều kiện:} Biến trạng thái \texttt{sysState} được cập nhật chính xác, kích hoạt điều khiển đồng bộ cho UC03, UC04 và UC05.
\end{itemize}

\subsubsection{UC03: Tự Động Điều Khiển Máy Bơm Thoát Nước}
\begin{itemize}
    \item \textbf{Mã định danh:} \texttt{UC03}
    \item \textbf{Tác nhân:}
    \begin{itemize}
        \item \textit{Tác nhân chính:} Trình điều phối FSM.
        \item \textit{Tác nhân phụ:} \texttt{«device»} Module Relay 5V \& Máy bơm mini.
    \end{itemize}
    \item \textbf{Mục tiêu:} Tự động kích hoạt máy bơm khi đạt ngưỡng nguy cơ và ngừng bơm dứt điểm khi nước đã rút về mức an toàn.
    \item \textbf{Tiền điều kiện:} Hệ thống ở chế độ tự động (\texttt{isAutoMode == true}).
    \item \textbf{Quan hệ \texttt{«include»}:} UC03 trực tiếp gọi:
    \begin{itemize}
        \item \textit{Kích Relay Active LOW:} Bật máy bơm khi FSM đạt mức DANGER hoặc CRITICAL.
        \item \textit{Thả nổi High-Z INPUT:} Triệt tiêu dòng rò optocoupler 5V, ngắt bơm hoàn toàn khi an toàn.
    \end{itemize}
    \item \textbf{Hậu điều kiện:} Máy bơm hoạt động hoặc ngắt nghỉ đúng quy trình an toàn.
\end{itemize}

    \item \textbf{Mã định danh:} \texttt{UC06}

    \end{itemize}
    \item \textbf{Mục tiêu:} Bảo vệ trạm tự động chống tràn bể khi người vận hành mất quyền giám sát từ xa hoặc rớt mạng Wi-Fi.

        \item \textit{Cưỡng chế đảo cờ isAutoMode = true:} Tự động thu hồi quyền thủ công, trao lại cho thuật toán FSM.
    \end{itemize}
    \item \textbf{Hậu điều kiện:} Trạm duy trì tính năng điều khiển tự trị (\textit{Autonomous Controller}), ngăn ngừa nguy cơ ngập úng khi người dùng bỏ quên chế độ thủ công.

\subsubsection{UC08: Giám Sát Trạng Thái Trạm Từ Xa Trên Giao Diện Web}
\begin{itemize}
    \item \textbf{Mã định danh:} \texttt{UC08}
    \item \textbf{Tác nhân chính:} Người vận hành (quan sát trực quan thông số).
    \item \textbf{Mục tiêu:} Hiển thị trực quan dữ liệu đo đạc và trạng thái phần cứng trên máy tính/điện thoại mà không cần màn hình LCD vật lý tại trạm.
    \item \textbf{Tiền điều kiện:} Thiết bị người dùng đã kết nối vào mạng Wi-Fi của trạm (\texttt{FloodGuard\_AP}) hoặc cùng Router Wi-Fi.
    \item \textbf{Quan hệ \texttt{«include»}:} UC08 trực tiếp gọi:
    \begin{itemize}
        \item \textit{Hiển thị đồng hồ mực nước \% \& khoảng cách:} Biểu diễn trực quan dạng vòng đo và thanh tiến trình.
