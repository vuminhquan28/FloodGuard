\chapter{Giới Thiệu Đề Tài \& Phân Tích Yêu Cầu}
\label{ch:introduction}

\section{Đặt Vấn Đề \& Khảo Sát Hiện Trạng}

\subsection{Bối Cảnh Thực Tiễn Về Ngập Úng Đô Thị}
Hiện tượng ngập úng đô thị cục bộ vào mùa mưa bão đang là một trong những thách thức dân sinh và hạ tầng nghiêm trọng nhất tại các đô thị lớn ở Việt Nam, điển hình như Hà Nội và TP. Hồ Chí Minh. Các điểm trũng thấp như hầm chui dân sinh (hầm chui Đại lộ Thăng Long, Nguyễn Trãi), tầng hầm các tòa nhà chung cư, trung tâm thương mại và các tuyến phố trũng thấp thường xuyên bị ngập sâu từ $0.5\,\text{m}$ đến hơn $1.2\,\text{m}$ chỉ sau những trận mưa rào kéo dài $30 - 45\,\text{phút}$. Tình trạng này gây thiệt hại nặng nề về phương tiện giao thông, làm tê liệt hạ tầng đô thị, gây chập cháy thiết bị điện tử ngầm và tiềm ẩn nguy cơ đe dọa trực tiếp tính mạng con người.

\subsection{Khảo Sát \& Đánh Giá Các Giải Pháp Hiện Nay}
Để ứng phó với ngập úng đô thị, các giải pháp giám sát và tiêu thoát nước hiện nay chủ yếu bao gồm:
\begin{itemize}
    \item \textbf{Cơ chế phao cơ truyền thống:} Sử dụng phao quả nổi đóng ngắt tiếp điểm điện cơ khí. Giải pháp này có ưu điểm chi phí thấp, tuy nhiên chỉ cung cấp thông tin nhị phân (ngập / cạn), không đo lường được tốc độ dâng nước, rất dễ kẹt rác cơ học và hoàn toàn không có khả năng giám sát hay cảnh báo sớm từ xa.
    \item \textbf{Hệ thống SCADA / PLC công nghiệp:} Sử dụng các bộ điều khiển logic khả trình (PLC) kết hợp cảm biến radar công nghiệp và mạng truyền thông cáp quang. Mặc dù có độ tin cậy và độ bền công nghiệp rất cao, giải pháp này có chi phí đầu tư phần cứng và bản quyền phần mềm cực kỳ đắt đỏ (lên tới hàng trăm triệu đồng cho một điểm đo), kích thước cồng kềnh, tiêu thụ nhiều năng lượng và khó có thể phủ kín diện rộng tại hàng trăm điểm ngập nhỏ lẻ trong đô thị.
    \item \textbf{Hệ thống IoT nhúng dựa trên vi điều khiển:} Đây là hướng tiếp cận hiện đại và tối ưu về mặt kinh tế kỹ thuật. Bằng việc ứng dụng các vi điều khiển thế hệ mới như ESP32 tích hợp sẵn kết nối không dây, hệ thống có thể thu nhỏ kích thước, tiết kiệm năng lượng, đo đạc thông số liên tục theo thời gian thực và cung cấp giao diện giám sát tiện lợi với chi phí phần cứng rất thấp.
\end{itemize}

\subsection{Phạm Vi \& Quy Mô Nghiên Cứu Của Đề Tài}
Trong khuôn khổ học phần \textit{Lập trình nhúng và thời gian thực}, đề tài tập trung nghiên cứu, thiết kế và chế tạo một \textbf{mô hình thực nghiệm quy mô trạm quan trắc (Proof of Concept / Scale Model)} mang tên \textbf{FloodGuard}. Mô hình mô phỏng khép kín toàn bộ chu trình vận hành của một trạm bơm tiêu thoát lũ thông minh: từ khâu đo đạc mực nước và lượng mưa, tiền xử lý tín hiệu số, thuật toán điều khiển máy trạng thái hữu hạn (FSM) kết hợp dải trễ, điều khiển cơ cấu chấp hành (bơm, van), cảnh báo đa giác quan tại trạm (LED, còi), đến giao diện giám sát và can thiệp điều khiển qua mạng không dây cục bộ.

\section{Mục Tiêu Kỹ Thuật Định Lượng Của Dự Án}
\label{sec:quantitative_goals}
Hệ thống \textbf{FloodGuard} hướng tới các chỉ tiêu kỹ thuật định lượng cụ thể:
\begin{enumerate}
    \item \textbf{Độ chính xác đo lường mực nước:} Sai số tuyệt đối đo khoảng cách qua cảm biến siêu âm $\le \pm 1.0\,\text{cm}$ trong dải đo làm việc thiết kế ($5 - 50\,\text{cm}$ trên cấu hình mô phỏng/bể chuẩn và $6 - 16\,\text{cm}$ trên mô hình demo mica thu nhỏ); độ phân giải hiển thị mức nước $1\%$ ($0\% - 100\%$).
    \item \textbf{Độ phân giải giám sát lượng mưa:} Thu nhận tín hiệu tương tự (ADC) 12-bit phân giải $\approx 0.8\,\text{mV}$ trên dải $0 - 3.3\,\text{V}$, phân loại 4 vùng trạng thái thời tiết qua 3 mốc ngưỡng ($3200 / 1800 / 1100$): Khô ráo, Mưa nhỏ, Mưa to và Ngập cảm biến.
    \item \textbf{Tính thời gian thực của thuật toán điều khiển (Worst-Case Reaction Time - WCRT):} Thời gian phản ứng tức thời từ lúc mực nước vượt ngưỡng nguy cấp đến khi chân GPIO xuất tín hiệu đóng Relay máy bơm khẩn cấp ($T_{\text{WCRT\_urgent}}$) không vượt quá $250\,\text{ms}$ (tính toán danh định đạt $203\,\text{ms}$ ở chu kỳ trích mẫu $T_{\text{sample}} = 150\,\text{ms}$). Với các chuyển trạng thái bậc thang thông thường có áp dụng bộ lọc chống rung thời gian $T_{\text{debounce}} = 1.0\,\text{s}$, thời gian ổn định là $T_{\text{stable}} \approx 1.2\,\text{s}$.
    \item \textbf{Chống rung giật cơ cấu chấp hành (Hysteresis \& Debounce Guard):} Áp dụng dải trễ Hysteresis biên độ $3\%$ mực nước ($41\%/38\%$, $71\%/68\%$, $21\%/18\%$) kết hợp bộ lọc chống rung thời gian Debounce $1.0\,\text{s}$ và thời gian chạy tối thiểu của máy bơm $5.0\,\text{s}$ (Min-run Time), đảm bảo triệt tiêu hoàn toàn hiện tượng đóng/ngắt Relay liên tục (chattering) do sóng nước dập dềnh quanh biên ngưỡng.
    \item \textbf{Cảnh báo trực quan \& âm thanh đa cấp độ:} Cụm 3 đèn LED phản ánh trực tiếp 4 trạng thái vận hành; còi Active Buzzer $5\,\text{V}$ phát xung âm thanh theo cơ chế phi chặn (\textit{Non-blocking}) qua hàm \texttt{millis()} ở tần số cộng hưởng nội $\approx 2.3\,\text{kHz}$ với 4 nhịp cảnh báo: Tắt (NORMAL) $\to$ Bíp thưa chu kỳ $1500\,\text{ms}$ (WARNING) $\to$ Bíp dồn dập chu kỳ $400\,\text{ms}$ (DANGER) $\to$ Hú liên tục (CRITICAL).
    \item \textbf{Hiệu năng giao tiếp mạng Web Dashboard:} Chu kỳ cập nhật dữ liệu telemetry thời gian thực $1000\,\text{ms}$ ($1\,\text{s}$), độ trễ phản hồi API cục bộ $8 - 18\,\text{ms}$, tỷ lệ phục vụ thành công các yêu cầu API đạt $\ge 99.5\%$ trên mạng Wi-Fi cục bộ ở cự ly $10\,\text{m}$.
    \item \textbf{Cơ chế bảo vệ an toàn tự trị (Failsafe Timeout):} Khi ở chế độ thủ công (\texttt{MANUAL}), nếu mất kết nối điều khiển mạng quá $30\,\text{s}$, trạm tự động thu hồi quyền thủ công và chuyển về chế độ tự động (\texttt{AUTO}) để bảo vệ công trình chống tràn bể; đồng thời kích hoạt quy tắc ghi đè an toàn tối cao (\textit{Safety Override}) nếu nước đạt mức CRITICAL quá $10\,\text{s}$.
\end{enumerate}

\section{Mô Hình Hóa Use Case Hệ Thống Theo UML}

\subsection{Phân Định Ranh Giới Hệ Thống \& Danh Sách Tác Nhân}
Mô hình hóa Use Case cho hệ thống FloodGuard được phân định rõ ràng giữa ranh giới phần mềm/phần cứng hệ thống và các tác nhân tương tác từ môi trường bên ngoài:
\begin{itemize}
    \item \textbf{Ranh giới hệ thống (\textit{System Boundary}):} Mang tên \textbf{Hệ Thống Trạm FloodGuard} (\textit{FloodGuard System}). Ranh giới bao trọn toàn bộ khối phần mềm nhúng thực thi trên vi điều khiển ESP32 bao gồm: tác vụ thu thập cảm biến, thuật toán FSM, tác vụ tạo xung cảnh báo, máy chủ HTTP Web Server và cơ chế giám sát an toàn Failsafe. Các thành phần logic nội bộ (như bộ điều phối FSM hay Web Server) nằm hoàn toàn trong ranh giới và không đóng vai trò là tác nhân.
    \item \textbf{Tác nhân chính (\textit{Primary Actors}):}
    \begin{enumerate}
        \item \textbf{Bộ Định Thời Timer / Watchdog (\textit{Internal Timer}):} Đóng vai trò là nguồn kích hoạt sự kiện thời gian thực định kỳ từ bên ngoài luồng logic chính, bao gồm bộ định thời chu kỳ đọc cảm biến $150\,\text{ms}$, bộ định thời chu kỳ thực thi FSM, và bộ đếm thời gian giám sát mất kết nối Failsafe ($30\,\text{s}$).
        \item \textbf{Người Vận Hành (\textit{Human Operator}):} Nhân sự giám sát và vận hành tương tác trực tiếp với trạm từ thiết bị đầu cuối (điện thoại, máy tính) thông qua trình duyệt Web.
    \end{enumerate}
    \item \textbf{Tác nhân phụ / Thiết bị ngoại vi (\textit{Secondary Actors / Devices}):}
    \begin{enumerate}
        \item \textbf{\texttt{«device»} Cảm biến siêu âm HC-SR04 \& Cảm biến mưa FC-37:} Thiết bị đầu vào cung cấp tín hiệu xung thời gian phản hồi (Echo) và điện áp tương tự (AO) / mức logic (DO).
        \item \textbf{\texttt{«device»} Module Relay 5V \& Máy bơm mini:} Cơ cấu chấp hành động lực tiếp nhận tín hiệu kích mức logic để đóng/ngắt nguồn động cơ bơm.
        \item \textbf{\texttt{«device»} Động cơ Servo SG90:} Cơ cấu chấp hành cơ khí tiếp nhận tín hiệu điều chế độ rộng xung (PWM) để định vị góc mở cửa van xả lũ ($0^\circ, 90^\circ, 180^\circ$).
        \item \textbf{\texttt{«device»} Cụm 3 Đèn LED \& Còi Buzzer 5V:} Thiết bị chỉ thị tại chỗ tiếp nhận tín hiệu logic và xung nhịp cảnh báo.
        \item \textbf{Người Dân Quanh Trạm (\textit{Human Actor}):} Tác nhân thụ hưởng tiếp nhận tín hiệu âm thanh còi hú và ánh sáng đèn LED cảnh báo nguy hiểm tại thực địa trạm.
    \end{enumerate}
\end{itemize}

\subsection{Sơ Đồ Use Case Tổng Thể Hệ Thống \& Đặc Tả Kiến Trúc Mô Hình}
\label{sec:uc_overall}
Mô hình Use Case tổng thể của trạm FloodGuard được tổ chức khoa học nhằm phân định rõ ràng giữa ranh giới hệ thống phần mềm nhúng và môi trường tương tác bên ngoài. Sơ đồ phân tách thành hai phân hệ chức năng độc lập về mặt logic nhưng phối hợp đồng bộ thông qua bộ nhớ trạng thái chung (Hình~\ref{fig:uc_overall}):
\begin{enumerate}
    \item \textbf{Phân hệ Giám sát \& Điều khiển Tự động Tại Chỗ (\textit{Autonomous Core Subsystem}):} Bao gồm 6 ca sử dụng cốt lõi (từ \texttt{UC01} đến \texttt{UC06}). Phân hệ này chịu sự kích hoạt định kỳ từ \textbf{Bộ Định Thời Timer / Watchdog} và các tín hiệu vật lý từ \textbf{Cảm biến siêu âm, Cảm biến mưa}. Mọi tác vụ từ thu thập dữ liệu, đánh giá máy trạng thái hữu hạn (FSM), xử lý dải trễ Hysteresis, điều khiển cơ cấu chấp hành (bơm, van) và phát cảnh báo tại trạm (LED, còi) đều vận hành hoàn toàn tự trị theo thời gian thực ($150\,\text{ms}$), không phụ thuộc vào tình trạng kết nối mạng Wi-Fi.
    \item \textbf{Phân hệ Giám sát \& Điều khiển Từ Xa Qua Web (\textit{Remote Web Telemetry Subsystem}):} Bao gồm 4 ca sử dụng (từ \texttt{UC07} đến \texttt{UC10}). Phân hệ này chịu sự kích hoạt trực tiếp từ \textbf{Người Vận Hành} thông qua trình duyệt Web. Người dùng có thể giám sát dữ liệu đo thời gian thực, chuyển đổi linh hoạt giữa chế độ \texttt{AUTO} và \texttt{MANUAL}, đồng thời can thiệp cưỡng chế trạng thái bật/tắt máy bơm và góc mở van xả.
\end{enumerate}

Toàn bộ đường truyền liên kết giữa các tác nhân và ca sử dụng được quy hoạch theo dạng trục xương cá (\textit{Bus Spine Line}), loại bỏ hoàn toàn hiện tượng giao cắt chéo rối mắt, đảm bảo chuẩn mực trực quan cao nhất của ngôn ngữ UML.

\section{Yêu Cầu Phi Chức Năng \& Ràng Buộc Thời Gian Thực}

\subsection{Đặc Tính \& Ràng Buộc Thời Gian Thực (Real-Time Constraints)}
Trong kỹ thuật lập trình hệ thống nhúng, FloodGuard được thiết kế theo mô hình hệ thống thời gian thực mềm (\textit{Soft Real-Time}) kết hợp với một số ràng buộc an toàn cứng (\textit{Firm Safety Constraints}):
\begin{itemize}
    \item \textbf{Chu kỳ trích mẫu cảm biến ($T_{\text{sample}} = 150\,\text{ms}$):} Vi điều khiển kích hoạt chu kỳ lấy mẫu định kỳ $150\,\text{ms}$. Trong các ứng dụng đo sâu công nghiệp, khoảng nghỉ khuyến cáo giữa các lần phát siêu âm thường $\ge 60\,\text{ms}$ để sóng dội tắt hẳn; tuy nhiên với mô hình sa bàn bể nông ($5 - 50\,\text{cm}$ hoặc mica $6 - 16\,\text{cm}$), hành trình khứ hồi của sóng âm chỉ mất dưới $3\,\text{ms}$, do đó việc đo 3 mẫu cách nhau $12\,\text{ms}$ trong chu kỳ quét $150\,\text{ms}$ hoàn toàn đảm bảo sóng âm tắt hẳn mà vẫn duy trì tính thời gian thực cao.
    \item \textbf{Bộ lọc cửa sổ trượt trung vị 3 mẫu (Median Filter):} Ba mẫu đo liên tiếp được lưu trong bộ đệm trượt để lọc bỏ các giá trị dị biệt (outliers) sinh ra do bọt khí, rung chấn cơ học hoặc bề mặt nước dập dềnh.
    \item \textbf{Thời gian phản ứng tối đa từ cảm biến đến chấp hành (WCRT $\le 250\,\text{ms}$):} Kể từ thời điểm cảm biến thu nhận xung Echo phản ánh mực nước vượt ngưỡng nguy cấp, thời gian phản ứng tức thời để thuật toán FSM ra quyết định và tiếp điểm Relay máy bơm đóng kín hoàn toàn ($T_{\text{WCRT\_urgent}}$) không vượt quá $250\,\text{ms}$ (tính toán đạt $203\,\text{ms}$). Đối với các chuyển trạng thái bậc thang thông thường, hệ thống áp dụng thêm bộ lọc chống rung thời gian Debounce $1.0\,\text{s}$ để đảm bảo độ tin cậy cơ cấu.
    \item \textbf{Chu kỳ làm mới giao diện Web ($T_{\text{refresh}} = 1000\,\text{ms}$):} Chu kỳ polling $1\,\text{s}$ từ Dashboard đảm bảo cân bằng tối ưu giữa độ tươi mới của dữ liệu hiển thị và tải bộ nhớ RAM/băng thông mạng của chip vi điều khiển ESP32.
\end{itemize}

\subsection{Độ Tin Cậy, Watchdog \& An Toàn Vận Hành (Safety \& Reliability)}
\begin{itemize}
    \item \textbf{Bộ giám sát tác vụ Task Watchdog Timer (TWDT):} Hệ thống cấu hình bộ định thời giám sát tác vụ phần mềm thuộc hệ điều hành ESP-IDF FreeRTOS (thông qua hàm \lstinline|esp_task_wdt_add(NULL)| gán trực tiếp cho \texttt{loopTask}) với chu kỳ timeout $3000\,\text{ms}$ ($3\,\text{s}$). Nếu vòng lặp chính của chương trình bị treo (do bẫy lỗi hoặc vòng lặp vô hạn), cơ chế TWDT sẽ tự động kích hoạt ngắt Reset vi điều khiển để khôi phục trạm về trạng thái an toàn ban đầu.
    \item \textbf{Cơ chế Failsafe mạng ($30\,\text{s}$):} Bộ đếm thời gian tầng ứng dụng đo khoảng cách giữa các gói tin điều khiển Web, tự động chuyển về AUTO nếu mất kết nối quá $30\,\text{s}$, ngăn ngừa rủi ro bỏ quên trạm ở chế độ thủ công.
\end{itemize}

\subsection{Phân Tích An Ninh \& Giới Hạn Bảo Mật (Security Analysis)}
Trong phạm vi mô hình thực nghiệm:
\begin{itemize}
    \item \textbf{Kiến trúc mạng:} Trạm phát sóng ở chế độ SoftAP (mạng \texttt{FloodGuard\_AP}) sử dụng chuẩn mã hóa WPA2-PSK yêu cầu mật khẩu để gia nhập mạng cục bộ.
    \item \textbf{Giới hạn hiện tại:} Máy chủ Web Server nhúng chạy trên giao thức HTTP cổng 80 không mã hóa (chưa hỗ trợ HTTPS/TLS) do chip ESP32 bị giới hạn về dung lượng RAM và mã hóa TLS tiêu tốn đáng kể chu kỳ xử lý CPU, gây ảnh hưởng đến tính thời gian thực của tác vụ điều khiển. Đồng thời, các endpoint API chưa áp dụng cơ chế xác thực Token hoặc OAuth2; các lệnh điều khiển hiện tại sử dụng phương thức \texttt{GET /api/control}.
    \item \textbf{Định hướng nâng cấp công nghiệp:} Đối với phiên bản triển khai thương mại diện rộng, hệ thống cần được nâng cấp: chuyển các lệnh thay đổi trạng thái sang phương thức chuẩn \texttt{POST /api/control} có kèm mã xác thực JSON Web Token (JWT), tích hợp vi mạch mã hóa bảo mật chuyên dụng (AtegCC608A) để xử lý HTTPS, và cấu hình danh sách trắng địa chỉ vật lý (MAC Address Whitelisting) chỉ cho phép các thiết bị chuyên trách được kết nối điều khiển trạm.
\end{itemize}
